import React from 'react';
import { StyleProp, StyleSheet, View, ViewStyle } from 'react-native';

import { colors } from '../../../branding/colors';
import { grid } from '../../../branding/grid';

export interface IComponentProps {
  children?: any;
  style?: StyleProp<ViewStyle>;
  testID?: string;
}

/**
 * Component
 *
 * @param style custom styling, overrides all other styles
 */
export default function Component({ style, children, testID = 'Component-test-id', ...props }: IComponentProps) {
  return (
    <View style={[baseStyle.view, style]} testID={testID}>
      {/* @ts-ignore */}
      {children}
    </View>
  );
}

const baseStyle = StyleSheet.create({
  view: {
    backgroundColor: colors.white,
    margin: grid.g8,
  },
});
